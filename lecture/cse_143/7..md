## LinkedList class

```java
public class LinkedIntList {
	private ListNode head;
	
	public void stutter() {
		ListNode curr = head;
		while (curr != null) {
			curr.next = new ListNode(curr.data, curr.next);
			curr = curr.next.next;
		}
	}
	
	public void add(int value) {
		if (head == null) {
			head = new ListNode(value);
		} else {
			ListNode curr = head;
			while (curr.next != null) {
				curr = curr.next;
			}
			
			curr.next = new ListNode(value);
		}
	}
}
```
