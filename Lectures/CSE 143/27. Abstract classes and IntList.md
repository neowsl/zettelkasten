```java
public abstract class AbstractIntList implements IntList {
	public void addAll(IntList other) {
		for (int n : other) {
			add(n);
		}
	}
	
	public void removeAll(IntList other) {
		Iterator<Integer> itr = iterator();
		while (itr.hasNext()) {
			int n = itr.next();
			if (other.contains(n)) {
				itr.remove();
			}
		}
	}
	
	public boolean isEmpty() {
		return !iterator().hasNext();
	}
	
	protected void checkIndex(int index) {
		if (index < 0 || index >= size()) {
			throw new IndexOutOfBoundsException();
		}
	}
	
	public Iterator<Integer> iterator() {
		return new ArrayIterator();
	}
	
	private class ArrayIterator implements Iterator<Integer> {
		// has a reference to outer AbstractIntList
		...
	}
}
```

A `static` inner class will not have a reference to the outer class.
